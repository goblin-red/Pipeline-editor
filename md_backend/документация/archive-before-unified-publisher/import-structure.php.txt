$count = dbTransaction(static function () use ($sections): int {
    $created = 0;
    $structureId = dbValue('SELECT id FROM documentation_pages WHERE slug = ?', ['memorandum-structure']);
    foreach ($sections as $slug => $page) {
        if (dbValue('SELECT id FROM documentation_pages WHERE slug = ? FOR UPDATE', [$slug])) continue;
        dbRun('INSERT INTO documentation_pages (parent_id, slug, title, summary_html, body_html, keywords, sort_order) VALUES (NULL, ?, ?, ?, ?, ?, ?)', [
            $slug, $page['title'], $page['summary'], $page['body'],
            json_encode($page['keywords'], JSON_UNESCAPED_UNICODE | JSON_THROW_ON_ERROR), $page['order'],
        ]);
        $pageId = dbId();
        if ($structureId) {
            dbRun('INSERT INTO documentation_links (source_page_id, target_page_id) VALUES (?, ?)', [$pageId, $structureId]);
            $body = $page['body'] . '<p>Место раздела в документации определено в [[' . dbId() . '|меморандуме о структуре документации]].</p>';
            dbRun('UPDATE documentation_pages SET body_html = ? WHERE id = ?', [$body, $pageId]);
        }
        $created++;
    }
    return $created;
});
echo "Создано основных рубрик: $count.\n";
