$count = dbTransaction(static function () use ($pages): int {
    $ids = [];
    foreach (dbAll('SELECT id, slug FROM documentation_pages') as $row) $ids[$row['slug']] = (int) $row['id'];
    $created = [];
    foreach ($pages as $slug => $page) {
        if (isset($ids[$slug])) continue;
        if ($page['parent'] !== null && !isset($ids[$page['parent']])) throw new RuntimeException('Не найден родитель: ' . $page['parent']);
        dbRun('INSERT INTO documentation_pages (parent_id, slug, title, summary_html, body_html, keywords, sort_order) VALUES (?, ?, ?, ?, ?, ?, ?)', [
            $page['parent'] !== null ? $ids[$page['parent']] : null, $slug, $page['title'], $page['summary'], $page['body'],
            json_encode($page['keywords'], JSON_UNESCAPED_UNICODE | JSON_THROW_ON_ERROR), $page['order'],
        ]);
        $ids[$slug] = dbId();
        $created[$slug] = $page;
    }
    foreach ($created as $slug => $page) {
        $order = 0;
        $replace = static function (array $match) use ($ids, $slug, &$order): string {
            $target = $match[1];
            if (!isset($ids[$target])) throw new RuntimeException('Не найден документ: ' . $target);
            $linkId = dbValue('SELECT id FROM documentation_links WHERE source_page_id = ? AND target_page_id = ?', [$ids[$slug], $ids[$target]]);
            if (!$linkId) {
                dbRun('INSERT INTO documentation_links (source_page_id, target_page_id, sort_order) VALUES (?, ?, ?)', [$ids[$slug], $ids[$target], $order++]);
                $linkId = dbId();
            }
            return '[[' . $linkId . '|' . $match[2] . ']]';
        };
        $summary = preg_replace_callback('/\[\[([a-z][a-z0-9-]*)\|([^\]]+)\]\]/u', $replace, $page['summary']);
        $body = preg_replace_callback('/\[\[([a-z][a-z0-9-]*)\|([^\]]+)\]\]/u', $replace, $page['body']);
        dbRun('UPDATE documentation_pages SET summary_html = ?, body_html = ? WHERE id = ?', [$summary, $body, $ids[$slug]]);
    }
    return count($created);
});
